cmake_minimum_required(VERSION 3.20)
project(SleepTimerWin32 VERSION 1.0 LANGUAGES CXX RC)

if(NOT WIN32)
    message(FATAL_ERROR "The native Sleep Timer requires Windows.")
endif()

configure_file(sleeptimer.rc.in sleeptimer.rc @ONLY)
add_executable(sleeptimer WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/sleeptimer.rc")
target_compile_features(sleeptimer PRIVATE cxx_std_17)
target_compile_definitions(sleeptimer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(sleeptimer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sleeptimer PRIVATE comctl32 powrprof ole32 uuid advapi32 shell32 user32)
if(MSVC)
    set_property(TARGET sleeptimer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(sleeptimer PRIVATE /W4 /permissive-)
    target_link_options(sleeptimer PRIVATE /MANIFEST:NO)
else()
    target_compile_options(sleeptimer PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(sleeptimer PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(timer_tests timer_tests.cpp)
    target_compile_features(timer_tests PRIVATE cxx_std_17)
    add_test(NAME timer_tests COMMAND timer_tests)
    add_executable(media_tests media_tests.cpp)
    target_compile_features(media_tests PRIVATE cxx_std_17)
    target_compile_definitions(media_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(media_tests PRIVATE user32)
    add_test(NAME media_tests COMMAND media_tests)
endif()
